# Policy: site images (photos, client logos) stay out of image search,
# but the site's own icons in /favicon/ are crawlable by everyone, so
# search engines and link previews can show them next to the site name.
# robots.txt never affects browsers: visitors always see every image.

# Dedicated image crawlers: favicon only.
# (Google fetches the search-result favicon as Googlebot-Image.)
User-agent: Googlebot-Image
Allow: /favicon/
Disallow: /

User-agent: Bingbot-Image
Allow: /favicon/
Disallow: /

# Link-preview fetchers: unrestricted, so shared links show icon and preview image.
User-agent: facebookexternalhit
User-agent: Twitterbot
Allow: /

# Everyone else: favicon allowed, other image files blocked.
User-agent: *
Allow: /favicon/
Allow: /favicon.ico
# Link-preview image (og:image) for LinkedIn, WhatsApp etc.
Allow: /img/easterbrook-logo.png
Disallow: /*.jpg$
Disallow: /*.jpeg$
Disallow: /*.png$
Disallow: /*.gif$
Disallow: /*.webp$
Disallow: /*.avif$
Disallow: /*.bmp$
Disallow: /*.svg$
Disallow: /*.ico$
